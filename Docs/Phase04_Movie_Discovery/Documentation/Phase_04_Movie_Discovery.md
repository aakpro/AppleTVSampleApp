// MARK: - Movie Discovery View Model (Combine-Based State Management)

class MovieDiscoveryViewModel: ObservableObject {
    // Published state channels for View binding
    @Published var movies: [Movie] = []  ✅
    @Published var isLoading: Bool = false  ✅
    @Published var errorMessage: String?  ✅
    
    // ViewModel Dependencies (injected via DI Container)
    private let repoFactory: APIRepositoryProtocol.Factory  
    private let useCaseFactory: GetMoviesUseCaseProtocol.Factory
    
    init(viewModelDependencies: ViewModelDependencies) {
        self.repoFactory = viewModelDependencies.apiRepo
        self.useCaseFactory = viewModelDependencies.useCases.getMovies
        
        // Initialize repository from factory (DI injection point)
        self.apiRepository = repoFactory.create() 
        
        // Wire Use Case Adapter → Repository (Application Layer pattern)
    }
    
    // Public API - Search/Filter Methods (Trigger UI events when search input changes)
    func performSearch(query: String, filterByGenre: [Int]?) async {
        isLoading = true
        errorMessage = nil
        
        do {
            try await useCaseFactory.execute(
                query: query,
                genresFilter: filterByGenre ?? []
            ).get() // Get results back to ViewModel

## 🔗 Application Layer Integration  

### Repository Abstraction + DI Container Wiring (`RepositoryIntegration.swift`)
```swift
// MARK: - Repository Injection Pattern (Application Layer)

class MovieDiscoveryIntegration: RepositoryProviderProtocol {
    
    // Dependency injection from DI container
    let movieRepositoryFactory = MOVIEDISCOVERY_API_REPOSITORY_FACTORY  
    let useCaseFactory = GET_MOVIES_USE_CASE_FACTORY
    
    init(injector: ApplicationDependencyInjector) throws {
        self.apiRepositoryFactory = .init(
            apiRepoFactory: injector.repoFactory,
            rateLimiter: Injector.rateLimitService.factory.create(),
            cacheManager: Injector.cacheServiceFactory.create()
        )  
        
        // Wire domain UseCase to repository abstraction layer
        self.useCaseAdapter = .init(
            repo: movieRepositoryFactory.create(),
            useCases: [getMoviesUseFactory()],
            domainLogic: Domain.GetMoviesDomain() 
        )
    }
}
```
