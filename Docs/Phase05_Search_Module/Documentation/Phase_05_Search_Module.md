## 📝 Files Generated by Phase 5 Autopilot

### Query Builder Protocol (`Querying/SearchBuilder.swift`):
```swift
// MARK: - Search Query Building Protocol (Clean Architecture Interface)

protocol SearchQueryBuilderProtocol {
    /// Build TMDB API-compatible search query with all parameters
    func buildSearchQuery(queryParams: String, filterParams: FilterParameters?) -> APIRequestParameters
    
    /// Execute actual TMDB search request
    func execute(search: TMSearchRequest, page: Int) async throws -> [TVSearchResult]
}

// MARK: - Implementation (Pure Swift builders, no external dependencies)
class SearchQueryBuilderImpl: SearchQueryBuilderProtocol {
    
    // Build API endpoint with filter parameters
    func buildSearchQuery(queryParams: String, filterParams: FilterParameters?) -> APIRequestParameters {
        return TMSearchRequest(
            url: "/search",
            query_params: queryParams,
            filtered_by_genres: filterParams?.genres.joined(separator: ","),
            page: filterParams?.page ?? 1
        )
    }
    
    // Execute with built parameters
    func execute(search: TMSearchRequest, page: Int) async throws -> [TVSearchResult] {  
        let params = buildRequest()
        return try await NetworkManager.request<TVSearchResult>(url: params.url)...
    }
}  
```

## 🔧 Debouncing Pattern (`Filters/Debouncer.swift`)

### Implementation Pattern (Combine Publishers):
```swift
// MARK: - Search Debouncer (300ms delay for UI input smoothing)

struct SearchDebouncer {
    private let debounceDelay: Double = 0.5 // 500ms
    
    func createSearchDebouncing(onQueryChange: String -> void @escaping () -> Void) 
        -> StringPublisher {
        
        return Just($0)  
            .debounce(for: .milliseconds(300), schedule: .immediate)
            .compactMap(String Publisher.init)
    
    // Example usage in ViewModel with SearchBarComponent integration
}

// MARK: - Usage Pattern (SearchViewModel.swift + Combined with API Layer)  
class SearchViewModel: ObservableObject {
    @Published var searchText: String = ""  
    private let debouncer = SearchDebouncer()
    private let queryBuilder = SearchQueryBuilderImpl()
    
    func onSearchTextChanged(_ newText: String) async {
        // Debounce the search request (only trigger after 300ms idle)  
        await withCheckedContinuation { continuation in
            self.debouncer.debounceDelay {
                do { 
                    let query = queryBuilder.build(queryParams: newText, filterParams: nil).get()
                    
                    continuation.resume(with: .init(result: .success(newText)))  
                
                } catch { 
                    continuation.resume(with: .init(result: .failure(error)))  
                }
            }
        }
    }
}
```
